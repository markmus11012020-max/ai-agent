    # ---------- транспорт ----------
    # Карта альтернативных имён методов Multitour API v2.
    # Если шлюз вернул "Method not found" — пробуем альтернативы.
    METHOD_ALIASES: Dict[str, List[str]] = {
        "GetDepartCities":  ["GetDepartureCities", "GetCities", "DepartCities"],
        "GetCountries":     ["GetCountryList", "Countries", "CountryList"],
        "GetResorts":       ["GetResortList", "Resorts"],
        "GetHotels":        ["GetHotelList", "Hotels"],
        "GetHotelStars":    ["GetHotelCategories", "GetStars", "GetHotelCategory"],
        "GetMeals":         ["GetMealTypes", "GetBoards", "GetMeal"],
        "GetTourOperators": ["GetOperators", "GetTourOperatorList"],
        "GetTourDates":     ["GetDates", "GetDepartureDates"],
        "CreateSearch":     ["CreateSearchRequest", "SearchCreate", "StartSearch"],
        "GetSearchState":   ["GetLoadState", "GetSearchStatus", "GetState"],
        "GetLoadState":     ["GetSearchState", "GetSearchStatus", "GetState"],
        "GetSearchResult":  ["GetResults", "GetTours", "GetSearchResults"],
        "ActualizePrice":   ["Actualize", "GetActualPrice", "UpdatePrice"],
        "Actualize":        ["ActualizePrice", "GetActualPrice"],
        "CreateOrder":      ["SaveTourOrder", "SaveOrder", "AddOrder"],
        "SaveOrder":        ["SaveTourOrder", "CreateOrder", "AddOrder"],
        "SaveTourOrder":    ["SaveOrder", "CreateOrder", "AddOrder"],
    }

    def _call(
        self,
        method: str,
        request: Optional[Dict[str, Any]] = None,
        _aliases: Optional[List[str]] = None,
    ) -> Dict[str, Any]:
        if not self.token:
            raise MultitourAPIError(
                "Не задан MULTITOUR_TOKEN (или API_KEY) в .env — "
                "получите токен в личном кабинете Multitour.ru."
            )
        tried: List[str] = []
        if _aliases:
            candidates = list(_aliases) + [method]
        elif method in self.METHOD_ALIASES:
            candidates = [method] + self.METHOD_ALIASES[method]
        else:
            candidates = [method]
        last_error: Optional[MultitourAPIError] = None
        for m in candidates:
            if m in tried:
                continue
            tried.append(m)
            payload = {
                "header": {"token": self.token, "method": m},
                "request": request or {},
            }
            try:
                resp = self.session.post(
                    self.base_url, json=payload, timeout=self.timeout
                )
            except requests.RequestException as exc:
                raise MultitourAPIError(f"Сетевая ошибка: {exc}") from exc
            if resp.status_code != 200:
                raise MultitourAPIError(
                    f"HTTP {resp.status_code}: {resp.text[:300]}"
                )
            try:
                data = resp.json()
            except ValueError as exc:
                raise MultitourAPIError(
                    f"JSON parse error: {exc}\n{resp.text[:300]}"
                ) from exc
            if data.get("is_success", False):
                return data.get("response") or {}
            errs = data.get("error") or []
            err_text = "; ".join(errs) if errs else "неизвестная ошибка"
            last_error = MultitourAPIError(f"Multitour: {err_text}")
            # Только при "Method not found" пробуем альтернативы.
            if not any("method" in e.lower() for e in errs):
                raise last_error
        raise last_error or MultitourAPIError(
            f"Multitour: Method not found ({method})"
        )
